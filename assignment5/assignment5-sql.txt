# Task 1
SELECT order_id, SUM(price * quantity) AS total_price 
FROM line_items 
JOIN products ON line_items.product_id = products.product_id 
GROUP BY order_id 
ORDER BY order_id ASC 
LIMIT 5;

# Task 2
SELECT c.customer_name, AVG(order_totals.total_price) AS average_order_price
FROM (
  SELECT o.customer_id, o.order_id, SUM(p.price * li.quantity) AS total_price
  FROM orders o
  JOIN line_items li ON o.order_id = li.order_id
  JOIN products p ON li.product_id = p.product_id
  GROUP BY o.customer_id, o.order_id
) AS order_totals
JOIN customers c ON order_totals.customer_id = c.customer_id
GROUP BY c.customer_id, c.customer_name
ORDER BY c.customer_name ASC;

SELECT first_name, last_name, COUNT(order_id) AS order_count
FROM employees
JOIN orders ON employees.employee_id = orders.employee_id
GROUP BY employees.employee_id, first_name, last_name
HAVING COUNT(order_id) > 5
ORDER BY last_name ASC;
