-- Task 1
SELECT order_id, SUM(price * quantity) AS total_price 
FROM line_items 
JOIN products ON line_items.product_id = products.product_id 
GROUP BY order_id 
ORDER BY order_id ASC 
LIMIT 5;

-- Task 2
SELECT customer_name, average_order_price 
FROM (
  SELECT customer_id, AVG(total_price) AS average_order_price 
  FROM orders 
  GROUP BY customer_id
) AS subquery
JOIN customers ON subquery.customer_id = customers.customer_id
ORDER BY customer_name ASC;

SELECT employees.employee_id, first_name, last_name 
FROM employees 
JOIN orders ON employees.employee_id = orders.employee_id 
GROUP BY employees.employee_id, first_name, last_name 
HAVING COUNT(order_id) > 5 
ORDER BY last_name ASC;
